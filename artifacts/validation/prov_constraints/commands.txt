--- Step 1: Check provconvert version ---

  $PROVCONVERT --version

--- Step 2: Convert PROV-JSON → PROV-N (required for validation) ---

  # Single-assessment Level-1 document

    -infile artifacts/paper/One_prov_output_enriched.json \
    -outfile artifacts/validation/prov_constraints/single_assessment.provn

  # Full itwinai Level-1 document (279 assessments)
  $PROVCONVERT \
    -infile Provenance_documents/interTwin-eu_itwinai_prov_output.json \
    -outfile artifacts/validation/prov_constraints/full_itwinai.provn

  # Level-2 comparison document (fc45...fd9c → 4086...54f4)
  $PROVCONVERT \
    -infile "Compare_commit_provenance/itwinai_commit_provenance_fc45...fd9c_to_4086...54f4.json" \
    -outfile artifacts/validation/prov_constraints/level2_fc45_4086.provn

--- Step 3: Run PROV-CONSTRAINTS validation ---

  # Validate each PROV-N file (provconvert checks PROV-CONSTRAINTS during conversion)
  $PROVCONVERT \
    -infile artifacts/validation/prov_constraints/single_assessment.provn \
    -outfile /dev/null \
    2>&1 | tee artifacts/validation/prov_constraints/single_assessment_validation.txt

  $PROVCONVERT \
    -infile artifacts/validation/prov_constraints/full_itwinai.provn \
    -outfile /dev/null \
    2>&1 | tee artifacts/validation/prov_constraints/full_itwinai_validation.txt

  $PROVCONVERT \
    -infile artifacts/validation/prov_constraints/level2_fc45_4086.provn \
    -outfile /dev/null \
    2>&1 | tee artifacts/validation/prov_constraints/level2_fc45_4086_validation.txt
